        May need the minigpt4/datasets/datasets/object_point_dataset.py to have this modified function:

        import open3d as o3d

            def _load_objaverse_point_cloud(self, object_id):
                #@@@orig
                # filename = f"{object_id}_{self.pointnum}.npy"
                #@@@new
                if '@' in object_id:
                    filename = f"{object_id}.ply"
                else:
                    filename = f"{object_id}_{self.pointnum}.npy"

                # point_cloud = np.load(os.path.join(self.data_path, filename)) #@@@orig
                #@@@new
                pcd = o3d.io.read_point_cloud(os.path.join(self.data_path, filename))
                xyz = np.asarray(pcd.points)
                if pcd.has_colors():
                    colors = np.asarray(pcd.colors)
                else:
                    colors = np.zeros_like(xyz)
                # Stack to get (N, 6): x, y, z, r, g, b
                pc6 = np.hstack([xyz, colors])
                target_n = 8192
                if pc6.shape[0] >= target_n:
                    idx = np.random.choice(pc6.shape[0], target_n, replace=False)
                    pc6 = pc6[idx]
                else: #Just add points of zeros to the end
                    pad = np.zeros((target_n - pc6.shape[0], 6), dtype=pc6.dtype)
                    pc6 = np.vstack([pc6, pad])
                point_cloud = pc6

                if not self.use_color:
                    point_cloud = point_cloud[:, :3]
                    #@@@new
                    raise Exception("@@@Color is being omitted; that should not happen!")

                return point_cloud
